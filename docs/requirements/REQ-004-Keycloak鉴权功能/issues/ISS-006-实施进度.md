  // 支持admin和super_user

    redirect('/admin/login');
  }

  // 根据角色过滤项目
  let projects;

    // 管理员: 显示所有项目
    projects = await db.projects.findMany({
      orderBy: { updatedAt: 'desc' }
    });

      where: { createdByUserId: 'admin' },
      orderBy: { updatedAt: 'desc' }
    });

  return <HomeClient
    projects={projects}
    role={session.isAdmin ? 'admin' : 'super_user'}
  />;

## 🐛 问题修复记录

### Bug修复1: 未登录用户访问管理员项目不显示登录按钮

**发现时间**: 2025-11-18
**修复时间**: 2025-11-18

**问题描述**:
用户报告: "UI测试发现问题,我没登录任何账号的情况下,访问由admin管理员创建的项目时,只有'无权访问此项目'的提示,并没有提示可以去登录super_user账号"

**根本原因**:
1. 未登录用户访问admin项目时,在authMiddleware步骤2(Token验证)就被拦截
2. 返回通用错误消息 "未授权访问,请重新登录"
3. 从未到达步骤3(权限检查),因此不会返回admin项目专用错误 "仅管理员和审核员可访问"
4. 前端layout.js仅检查 "仅管理员和审核员可访问" 来显示登录按钮
5. 结果: 未登录用户看不到登录按钮

**解决方案**:
在authMiddleware步骤2(Token验证)中,当检测到无Token时,先检查请求的项目是否是admin项目:
- 如果是admin项目 → 返回 "此项目仅管理员和审核员可访问"
- 如果是普通项目 → 返回 "未授权访问,请重新登录"

这样前端就能正确检测到admin项目并显示登录按钮。

**修改文件**:
1. `lib/custom/auth/middleware.js` (步骤2 Token验证部分,添加admin项目检测逻辑)
2. `app/projects/[projectId]/layout.js` (简化错误检测逻辑,移除不必要的projectInfo状态)

**测试结果**:
- ✅ 未登录访问admin项目: 返回 "此项目仅管理员和审核员可访问" (触发登录按钮)
- ✅ 未登录访问普通项目: 返回 "未授权访问,请重新登录" (不显示登录按钮)
- ✅ super_user访问admin项目: 正常访问,返回项目数据
- ✅ 所有原有测试场景不受影响

**修改文件**:
1. `app/api/projects/route.js` (+9行) - POST方法添加admin Session检查

### Bug修复3: errorHandler固定替换错误消息导致登录按钮不显示

**发现时间**: 2025-11-18
**修复时间**: 2025-11-18

**问题描述**:
用户报告: "新创建的admin项目,在无痕浏览器里访问还是'无权访问此项目'的提示,没有登录按钮"
- API返回: `{"error":"此项目仅管理员和审核员可访问"}` (正确)
- 前端显示: "无权访问此项目" (错误,被替换了)

**根本原因**:
1. layout.js设置 `allowLoginRedirect: false` (ISS-002安全优化,防止暴露admin登录入口)
2. errorHandler.js检测到 `allowLoginRedirect: false` 时,固定替换错误消息为"无权访问此项目"
3. 丢失了API返回的admin项目专用错误消息 "此项目仅管理员和审核员可访问"
4. 前端无法识别admin项目,不显示登录按钮

**解决方案**:
修改 `errorHandler.js` 401错误处理逻辑:
- 原逻辑: `allowLoginRedirect: false` 时,固定替换为"无权访问此项目"
- 新逻辑: `allowLoginRedirect: false` 时,**保留API返回的原始错误消息**
- 这样admin项目的专用错误消息能传递到前端,触发登录按钮显示

**修改文件**:
1. `lib/custom/utils/errorHandler.js` (第84-86行) - 保留原始错误消息

**测试结果**:
- ✅ 未登录访问admin项目: 前端显示 "此项目仅管理员和审核员可访问"
- ✅ 登录按钮正确显示
- ✅ 点击登录跳转到 `/super-user/login?redirect=...`
- ✅ ISS-002安全优化不受影响(仍然不会跳转到admin登录页)
